function abrirConvite() { const telaEnvelope = document.getElementById("tela-envelope"); const telaConvite = document.getElementById("tela-convite"); if (!telaEnvelope || !telaConvite) { return; } if (telaEnvelope.classList.contains("abrindo")) { return; } telaEnvelope.classList.add("abrindo"); setTimeout(function () { telaConvite.classList.add("visivel"); }, 450); setTimeout(function () { telaEnvelope.classList.add("saindo"); }, 1000); } /* ========================================== ABRIR O CONVITE DIRETAMENTE QUANDO VOLTAR DA LISTA DE PRESENTES ========================================== */ document.addEventListener("DOMContentLoaded", function () { const parametros = new URLSearchParams(window.location.search); const conviteAberto = parametros.get("convite") === "aberto"; if (!conviteAberto) { return; } const telaEnvelope = document.getElementById("tela-envelope"); const telaConvite = document.getElementById("tela-convite"); if (!telaEnvelope || !telaConvite) { return; } /* Esconde o envelope imediatamente */ telaEnvelope.style.display = "none"; /* Mostra o convite */ telaConvite.classList.add("visivel"); });
